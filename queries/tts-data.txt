--TTs query
SELECT "TTS:Help Desk".Description, "TTS:Help Desk".Assignee, "TTS:Help Desk"."Assigned Group", "TTS:Help Desk"."Incident Number", "TTS:Help Desk"."Submit Date", "TTS:Help Desk".Status, "TTS:Help Desk".Urgency, "TTS:Help Desk"."Categorization Tier 1"FROM "TTS:Help Desk" "TTS:Help Desk"
WHERE ("TTS:Help Desk".Status='Assigned') AND ("TTS:Help Desk".Urgency='1-Critical') AND ("TTS:Help Desk"."Categorization Tier 1"='DEGRADACION DE CALIDAD') OR ("TTS:Help Desk".Status='In Progress') AND ("TTS:Help Desk".Urgency='1-Critical') AND ("TTS:Help Desk"."Categorization Tier 1"='DEGRADACION DE CALIDAD') OR ("TTS:Help Desk".Status='Pending') AND ("TTS:Help Desk".Urgency='1-Critical') AND ("TTS:Help Desk"."Categorization Tier 1"='DEGRADACION DE CALIDAD')
ORDER BY "TTS:Help Desk"."Incident Number"
